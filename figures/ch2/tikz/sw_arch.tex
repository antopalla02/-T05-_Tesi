% Block diagram of the autonomous driving software (Section 2.4).
\begin{tikzpicture}[
    font=\footnotesize,
    >={Stealth[length=2mm]},
    blk/.style={draw, rounded corners=2pt, align=center, minimum height=8mm, inner sep=3pt},
    sens/.style={blk, fill=gray!8},
    proc/.style={blk, minimum width=20mm, fill=blue!6},
    mod/.style={blk, minimum width=18mm, fill=blue!6},
    hl/.style={blk, minimum width=18mm, fill=blue!18, very thick},
    grp/.style={draw, dashed, rounded corners=4pt, inner sep=4pt},
    lab/.style={font=\scriptsize, align=center, text=black!70},
    line/.style={->, thick},
    dot/.style={circle, fill, inner sep=0pt, minimum size=2.2pt}
]
% per-sensor processing (perception)
\node[proc] (p3) {LiDAR\\processing};
\node[proc, below=3mm of p3] (p4) {RaDAR\\processing};
\node[proc, below=3mm of p4] (p5) {Camera\\processing};

% target tracking, right of LiDAR, RaDAR and camera
\path (p3.north east) -- (p5.south east) coordinate[pos=0.5] (tmid);
\node[hl, anchor=west, minimum height=34mm] (trk) at ([xshift=18mm]tmid) {Target\\tracking};
\node[grp, fit=(p3)(p5)(trk), label={[lab, anchor=south west, inner sep=1pt]north west:Perception}] (P) {};

% estimation, above target tracking
\node[mod, minimum height=12mm, anchor=south] (est) at ([yshift=16mm]trk.north) {Estimation};

% sensors, from the height of estimation to the bottom of perception
\path ([xshift=-7mm]p3.west |- est.north) coordinate (s1) ([xshift=-21mm]p5.south west) coordinate (s2);
\node[sens, fit=(s1)(s2), inner sep=0pt] (sens) {};
\node[rotate=90, align=center] at (sens) {Exteroceptive and\\proprioceptive sensors};

% planning and control, aligned with tracking
\node[mod, anchor=west] (plan) at ([xshift=14mm]trk.east) {Planning};
\node[mod, right=14mm of plan] (ctrl) {Control};
\node[mod, fill=gray!8, below=7mm of plan] (db) {Track\\database};
\node[sens, minimum width=18mm, below=7mm of ctrl] (act) {Actuators};

% sensors to processing and to estimation
\foreach \p in {p3,p4,p5} \draw[line] (sens.east |- \p) -- (\p.west);
\draw[line] (sens.east |- est) -- (est.west);
\coordinate (bus) at ([xshift=4mm]p5.east);

% processing to tracking (LiDAR, RaDAR, camera)
\foreach \p in {p3,p4,p5} \draw[line] (\p.east) -- (\p.east -| trk.west);
\path (bus |- p4) -- (p4.east -| trk.west) node[lab, above, midway] {detections};

% estimation to tracking, planning and control
\draw[line] (est) -- node[lab, right, pos=0.4]{ego state} (trk);
\coordinate (eb1) at (est.east -| plan);
\coordinate (eb2) at (est.east -| ctrl);
\draw[thick] (est.east) -- (eb2);
\draw[line] (eb1) -- (plan);
\draw[line] (eb2) -- (ctrl);
\node[dot] at (eb1) {};
\node[lab, above] at ($(est.east)!0.5!(eb1)$) {ego state};

% planning and control
\draw[line] (trk.east |- plan) -- node[lab, above]{opponents} (plan);
\draw[line] (db) -- (plan);
\draw[line] (plan) -- node[lab, above]{reference\\trajectory} (ctrl);
\draw[line] (ctrl) -- node[lab, right]{steering, throttle,\\brake, gear} (act);

% vehicle, closing the loop between actuators and sensors
\node[inner sep=1pt, label={[lab]below:Vehicle}] (car) at ([yshift=-12mm]$(sens |- act)!0.5!(act)$) {\reflectbox{\includegraphics[width=30mm]{figures/ch2/src/eav24_top.png}}};
\draw[line] (act) |- (car);
\draw[line] (car) -| (sens.south);
\end{tikzpicture}
